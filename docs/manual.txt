NAME
  get_env(3f) - [M_unicode:QUERY] a function returning the value of an
  environment variable (LICENSE:PD)

SYNTAX
  function get_env(NAME,DEFAULT,IERR=IERR) result(VALUE)

      character(len=*),intent(in)	     :: NAME

      ! ONE OF ...
      character(len=*),intent(in),optional :: DEFAULT
      real,intent(in),optional		   :: DEFAULT
      integer,intent(in),optional	   :: DEFAULT
      doubleprecision,intent(in),optional  :: DEFAULT
      logical,intent(in),optional	   :: DEFAULT
      type(unicode_type),optional	   :: DEFAULT

      integer,intent(out),optional	     :: IERR

      ! ONE OF THE FOLLOWING, MATCHING TYPE OF DEFAULT
      character(len=:),allocatable	   :: VALUE
      integer				   :: VALUE
      real				   :: VALUE
      doubleprecision			   :: VALUE
      logical				   :: VALUE
      type(unicode_type)		   :: VALUE

DESCRIPTION
  Get the value of an environment variable or optionally return a default
  value when the environment variable is not set or is set to a blank string.

  The type returned is the same as the type of the default value.

OPTIONS
  NAME
    name of environment variable.

  DEFAULT
    value to return if environment variable is not set or set to an empty
    string. May be CHARACTER, REAL, INTEGER, LOGICAL, DOUBLEPRECISION or
    TYPE(UNICODE_TYPE.	Defaults to a null CHARACTER value.

RETURNS
  VALUE
    the value of the environment variable or the default.  The type is the
    same as DEFAULT unless NAME is CHARACTER but VALUE is DEFAULT, in which
    case TYPE(UNICODE_TYPE) is returned. That is, if either NAME or VALUE is
    TYPE(UNICODE_TYPE) the returned value is TYPE(UNICODE_TYPE). If an error
    occurs and it is numeric, huge(0|0.0|0.0d0) is returned.

    For a LOGICAL type, Any environment variable value starting with F,f,N or
    n is .FALSE. and any value starting with Y,y,T or t is true. A leading
    period (".") is ignored.  Anything else returns .false. .

  IERR
    return error code. Must be specified with a keyword.  It is zero if no
    error occurred.

glob(3m_unicode)					      glob(3m_unicode)

NAME
  glob(3f) - [M_unicode:COMPARE] compare given string for match to a pattern
  which may contain globbing wildcard characters (LICENSE:MIT)

SYNOPSIS
  logical function glob(string, pattern ) result (uline)

      type(unicode_type),intent(in) :: string
      ! or
      character(len=*),intent(in)   :: string

      type(unicode_type),intent(in) :: pattern
      ! or
      character(len=*),intent(in)   :: pattern

      logical			    :: uline

DESCRIPTION
  glob(3f) compares an (entire) STRING for a match to a PATTERN which may
  contain basic wildcard "globbing" characters.

  •  "*" matches any string.

  •  "?" matches any single character.

  •  a NULL may not appear in the input strings

  •  trailing whitespace is significant

  In this version to get a match the entire string must be described by
  PATTERN. Trailing whitespace is significant, so trim the input string if it
  is desired to ignore trailing whitespace.

  A NULL character is added to the input strings internally to avoid early
  matches. Without this, patterns like "b*ba" fail on a string like "babababa"
  because the first match found is not at the end of the string so 'baba' does
  not match 'babababa'. So the algorithm is said to find an early match.

  One method that allows skipping over the early matches is to insert an extra
  character at the end of the string and pattern that does not occur in the
  pattern. Typically a NULL is used (char(0)), as it is here. So searching for
  b*ba\0 in babababa\0 matches the entire string.

OPTIONS
  string
    the input string to be tested for a match to the pattern.

  pattern
    the globbing pattern to search for. The following simple globbing rules
    apply:

    •  "?" matching any one character

    •  "*" matching zero or more characters.  Do NOT use adjacent asterisks.

    •  The input strings must not contain a NULL character

    •  spaces are significant and must be matched.

    •  There is no escape character, so matching strings with a literal
       question mark and asterisk is problematic.

EXAMPLES
  Example program

      program demo_glob
      use M_unicode, only : glob, trim, unicode_type, len
      use M_unicode, only : expand_backslash
      use M_unicode, only : assignment(=)
      implicit none
      integer :: i
      type(unicode_type),allocatable :: ufiles(:)
      type(unicode_type),allocatable :: matched(:)
      character(len=*),parameter :: &
       filenames(*)= [character(len=256) :: &
       & 'My_favorite_file.F90',    & ! English
       & '我最喜欢的文档.c',	     & ! Mandarin_Chinese
       & 'मरी_पसदीदा_फाइल.f90',       & ! Hindu
       & 'Mi_archivo_favorito.c',   & ! Spanish
       & 'ملفي_المفضل.h',	  & ! Modern_Standard_Arabic
       & 'Mon_fichier_préféré.f90', & ! French
       & 'আমার_পরিয_ফাইল',	    & ! Bengali
       & 'Meu_arquivo_favorito',    & ! Portuguese
       & 'Мой_любимый_файл',	      & ! Russian
       & 'میری_پسندیدہ_فائل.pdf',   & ! Urdu
       & 'src/M_modules.F90',	    &
       & 'src/subset.inc',	    &
       & 'test/check.f90 ',	    &
       & 'app/main.f90 ']
      character(len=*),parameter :: &
       encoded(*)= [character(len=256) :: &
       & 'My_favorite_file.F90',		    & ! English
       & '\u6211\u6700\u559C\u6B22\u7684\u6587\u6863.c', & ! Mandarin_Chinese

       &\u092B\u093C\u093E\u0907\u0932.f90',	    & ! Hindu

       &\u0627\u0644\u0645\u0641\u0636\u0644.h ',   & ! Modern_Standard_Arabic
       & 'Mon_fichier_pr\xE9f\xE9r\xE9.f90',	    & ! French
       & '\u0986\u09AE\u09BE\u09B0_\u09AA\u09CD\u09B0\u09BF\u09AF\u09BC_&
       &\u09AB\u09BE\u0987\u09B2',		    & ! Bengali
       & 'Meu_arquivo_favorito',		    & ! Portuguese
       & '\u041C\u043E\u0439_\u043B\u044E\u0431\u0438\u043C\u044B\u0439_&
       &\u0444\u0430\u0439\u043B',		    & ! Russian
       & '\u0645\u06CC\u0631\u06CC_&
       &\u067E\u0633\u0646\u062F\u06CC\u062F\u06C1_&
       &\u0641\u0627\u0626\u0644.pdf',		    & ! Urdu
       & 'src/M_modules.F90', &
       & 'src/subset.inc', &
       & 'test/check.f90 ', &
       & 'app/main.f90 ']
      character(len=*),parameter :: &
	  g='(*(g0))', g1='(*(g0,1x))', comma='(*(g0:,", ",/))'

	 ! some basic usage
	 write(*,g)merge('PASSED','FAILED',glob("mississipPI", "*issip*PI"))
	 write(*,g)merge('PASSED','FAILED',glob("bLah", "bL?h"))
	 write(*,g)merge('PASSED','FAILED',glob("bLaH", "?LaH"))

	 ! create a list of trimmed filenames
	 ufiles=unicode_type(filenames)
	 ufiles=trim(ufiles)
	 write(*,g)'FILENAMES:'

      subroutine show_filenames(names)

      subroutine oop()
      use M_unicode, only : glob,ut=>unicode_type

      use M_unicode, only : assignment(=)
      use M_unicode, only : operator(//)

      !
      ! ignoring ς for simplicity
      USTRING='ΑαΒβΓγΔδΕεΖζΗηΘθΙιΚκΛλΜμΝνΞξΟοΠπΡρΣσςΤτΥυΦφΧχΨψΩω'

      print u,' string is : ' // USTRING
      ! pattern may be UTF-8 or ASCII
      print *,merge('PASSED','FAILED',USTRING%glob('*Α*Ζ*Ω*'))

      ! pattern may be unicode_type

      end subroutine oop
      end program demo_glob

NAME
  ICHAR(3f) - [M_unicode:CONVERSION] character-to-integer Unicode codepoint
  conversion function (LICENSE:MIT)

SYNOPSIS
  result = ichar(c)

      elemental integer function ichar(c,kind)

       type(unicode_type),intent(in) :: c

CHARACTERISTICS
  • c is a scalar character

  • the return value is of default integer kind.

DESCRIPTION
  This overload of ichar(3f) only returns values from input type

  unicode_type.
    This is because the intrinsic ichar(3f) returns the code for the character
    in the system's native character set. the correspondence between
    characters and their codes is not necessarily the same across different
    Fortran implementations. For example, a platform using EBCDIC would return
    different values than an ASCII platform.

  See IACHAR(3f) for specifically working with the ASCII character set.

OPTIONS
  •  C : The input character to determine the Unicode codepoint of.

RESULT
  The codepoint in the Unicode character set for the character being queried
  is returned.

  The result is the position of C in the Unicode collating sequence, which is
  generally not the dictionary order in a particular language.

  It is nonnegative and less than n, where n is the largest value in the
  Unicode collating sequence (ie. 1 114 111,).

  For any characters C and D capable of representation in the processor, C <=
  D is true if and only if ICHAR(C) <= ICHAR(D) is true and C == D is true if
  and only if ICHAR(C) == ICHAR(D) is true.

EXAMPLES
  sample program:

     program demo_ichar
     use M_unicode, only : assignment(=),ch=>character
     use M_unicode, only : ut=>unicode_type, write(formatted)
     use M_unicode, only : ichar, expand_backslash, len
     use M_unicode, only : ord=>ichar ! a common alias
     implicit none
     type(ut)		  :: string
     type(ut),allocatable :: lets(:)
     integer,allocatable  :: ilets(:)
     integer		  :: i
	!
	! create a string containing multibyte characters
	string=[949, 8021, 961, 951, 954, 945, 33] ! eureka
	write(*,'(*(DT,1x,"(AKA. eureka!)"))')string
	!
	! call ichar(3f) on each glyph of the string to convert
	! the string to an array of integer codepoints
	ilets=[(ichar(string%sub(i,i)),i=1,len(string))]
	write(*,'(*(z0,1x))')ilets
	!
	! note that the %codepoint method is commonly used to
	! convert a string to an integer array of codepoints
	write(*,'(*(z0,1x))')string%codepoint()
	!
	! it is important to remember ICHAR(3f) with an ASCII argument
	! is not overloaded so multibyte quoted characters will not
	! produce the correct unicode value, only unicode_type variables
	! will, unless the compiler has an extension (eg. currently
	! flang(1) will return 8364 but gfortran(1) will return 226) for
	! the first call.
	!write(*,'(a)')ichar('€')	 !<== INCORRECT
	write(*,'(a)')ichar(ut('€'))	 !<== CORRECT

	! elemental
	write(*,'("WRITING ISSUES:")')
	!
	! define an array LETS with escape codes with one glyph per element
	lets=[ut('\U03B5'),ut('\U1F55'),ut('\U03C1'),ut('\U03B7'), &
	    & ut('\U03BA'),ut('\U03B1'),ut('\U0021')]
	lets=expand_backslash(lets) ! convert escape codes to glyphs
	!
	! look at issues with converting to CHARACTER for simple printing
	!
	write(*,'("each element is a single glyph ",*(g0,1x))')len(lets)
	!
	! notice if you convert to an array of intrinsic CHARACTER type the
	! strings are all the same length in bytes; but unicode characters
	! can take various numbers of bytes
	write(*,'(*(g0,":"))')'CHARACTER array elements have same length',&
	   & len(ch(lets))
	!
	! this will not appear correctly because all elements are padded to
	! the same length in bytes
	write(*,'(*(a,":"))')ch(lets)
	! one element at a time will retain the size of each element
	write(*,'(*(a,":"))')(ch(lets(i:i)),i=1,size(lets))
	!
	! the FIRST LETTER of each element is converted to a codepoint so
	! for the special case where each string element is a single glyph
	! an elemental approach works
	write(*,'("ELEMENTAL:",*(z0,1x))')ichar(lets)

	! OOPS
	write(*,'("OOPS:     ",*(z0,1x))')lets%ichar()
     end program demo_ichar

NAME
  LOWER(3f) - [M_unicode:CASE] changes a string to lowercase over specified
  range (LICENSE:MIT)

SYNOPSIS
  pure elemental function lower(str) result (string)

       type(unicode_type),intent(in) :: str
       type(unicode_type)	     :: string

DESCRIPTION
  lower(str) returns a copy of the input string with all characters converted
  to miniscule (ie. "lowercase").

OPTIONS
  str
    string to convert to miniscule

RETURNS
  lower
    copy of the entire input string with all characters converted to
    miniscule.

TRIVIA
  The terms "uppercase" and "lowercase" date back to the early days of the
  mechanical printing press. Individual metal alloy casts of each needed
  letter or punctuation symbol were meticulously added to a press block, by
  hand, before rolling out copies of a page. These metal casts were stored and
  organized in wooden cases. The more-often-needed miniscule letters were
  placed closer to hand, in the lower cases of the work bench. The less often
  needed, capitalized, majuscule letters, ended up in the harder to reach
  upper cases.

     use M_unicode,	  only : ut => unicode_type, operator(==)

     character(len=*),parameter :: g='(*(g0))'
     type(unicode_type) :: pangram
     type(unicode_type) :: diacritics
     type(unicode_type) :: expected

       ! a sentence containing every letter of the English alphabet
       pangram="THE QUICK BROWN FOX JUMPS OVER THE LAZY DOG"
       expected="the quick brown fox jumps over the lazy dog"
       call test(pangram,expected)
       !
       ! Slovak pangram
       PANGRAM	  = 'VYPÄTÁ DCÉRA GRÓFA MAXWELLA S IQ NIŽŠÍM AKO &
       &KÔŇ NÚTI ČEĽAĎ HRÝZŤ HŔBU JABĹK.'
       expected   = 'vypätá dcéra grófa maxwella s iq nižším ako &
       &kôň núti čeľaď hrýzť hŕbu jabĺk.'
       call test(pangram,expected)
       !
       ! contains each special Czech letter with diacritics exactly once
       DIACRITICS='PŘÍLIŠ ŽLUŤOUČKÝ KŮŇ ÚPĚL ĎÁBELSKÉ ÓDY.'
       expected ='příliš žluťoučký kůň úpěl ďábelské ódy.'
       print g,'("A horse that was too yellow-ish moaned devilish odes")'
       call test(diacritics,expected)
     contains
     subroutine test(in,expected)
     type(unicode_type),intent(in) :: in
     type(unicode_type),intent(in) :: expected
     type(unicode_type) 	   :: lowercase
     character(len=*),parameter    :: nl=new_line('A')
	 write(stdout,g)in%character()
	 lowercase=lower(in)
	 write(stdout,g)lowercase%character()
	 write(stdout,g)merge('PASSED','FAILED',lowercase == expected ),nl

NAME
  M_unicode(3f) - [M_unicode::INTRO] Unicode string module (LICENSE:MIT)

DESCRIPTION
  The M_unicode(3f) module is a collection of Fortran string methods that work
  with UTF-8 encoded text not just ASCII-7 data.

  Strings are declared using the user-defined type "UNICODE_TYPE". The type
  supports allocatable ragged arrays where each element may be of differing
  length.

  Compiler support of the optional Fortran ISO_10646 extension is not
  required.

  The M_unicode(3f) module overloads the Fortran built-in CHARACTER intrinsics
  and operators to allow TYPE(UNICODE_TYPE) to use the intrinsic procedure
  names in much the same manner the intrinsics are used with CHARACTER
  variables.

  The intrinsic overloads include TOKENIZE(3f) and SPLIT(3f) even if the
  underlying compiler does not yet support those intrinsics.

  Overloads of assignment, logical comparisons, and concatenation using the //
  operator with strings are included as well to make use of TYPE(UNICODE_TYPE)
  largely consistent with standard CHARACTER string manipulations.

  Nearly all the methods are available using both OOP and procedural syntax.

  In addition M_unicode(3f) includes routines for parsing, tokenizing,
  changing case, substituting new strings for substrings, locating strings
  with simple wildcard expressions, removing tabs and line terminators and
  other advanced non-intrinsic string manipulations.

  The **UPPER()** and **LOWER()** functions support the concept of case for
  the Unicode Latin characters not just the ASCII subset, and a basic SORT()
  function provides for ordering the data by Unicode codepoint values.

  A PAD() function allows padding strings at least up to a specified glyph
  length with a repeating pattern.

  **M_unicode** should be useful for anyone working with UTF-8 data,
  particularly if the compiler does not support the UCS-4 extensions of
  Fortran.

  Until proven otherwise M_unicode(3f) should work with any environment where
  UTF-8 files are supported.

  The type components are not public to allow for use of the same user code
  when using other modules such as M_ucs4(3f) which ultimately will provide
  the same user interface but internally using ISO_10646 internal encoding
  instead of an array of integers containing codepoints (which is what
  M_unicode(3f) uses). This has the drawback of not permitting easy use of
  array syntax directly on the codepoint array. Perhaps this decision will
  change but in the meantime several methods such as REPLACE(3f) and
  CHARACTER(3f) and SUB(3f) provide similar functionality.

NAME
  M_unicode__eval(3f) - [M_unicode:QUERY] evaluate strings represented as
  Unicode codepoints (LICENSE:MIT)

SYNOPSIS
  Public Types

      public :: cp_string, var_value, user_func, user_func_iface, &
		& compiled_expr

  Public Procedures

      public :: compile_expr_cp, eval_compiled, &
	       & differentiate_compiled
      public :: utoc, ctou

DESCRIPTION
  M_unicode__eval is a standard-conforming Fortran 2008 module for compiling,
  evaluating, and symbolically differentiating mathematical expressions stored
  as 32-bit Unicode codepoint arrays (integer(int32)).

   CORE DESIGN PRINCIPLES
  •  Unicode-Native Representation: Expressions and variable names are
     represented as 1D arrays of integer(int32) Unicode scalar values,
     eliminating multi-byte UTF-8 indexing bugs.

  •  Strict Case-Sensitivity: Unlike standard Fortran, all identifiers (sin vs
     SIN, x vs X, α vs Α) are strictly case-sensitive.

  •  Compile-Once Bytecode VM: Expressions compile into flat stack-based
     bytecode (compiled_expr) for fast repeated evaluation inside numerical
     loops.

  •  Unified Scalar & Array Broadcasting: All variables are passed as
     var_value containers (val(:)). Scalars are 1-element arrays (size ==

     1)  and automatically broadcast against 1D arrays (size == N).

   EXPRESSION LANGUAGE REFERENCE
  Operators (Ordered Lowest to Highest Precedence)

	Precedence   Operators		      Description      Associativity
	1 (Lowest)   <, <=, ≤, >, >=,	      Comparisons (return    Left
		     ≥, ==, /=, ≠	      1.0 if true, 0.0 if
					      false)
	3	     *, ×, ·, /, ÷, implicit  Multiplication and     Left
					      division
	4	     +x, -x, −x, √x (U+221A)  Unary signs and	     Right
					      prefix square root
	5	     ^, **, ⁰–⁹, ⁺, ⁻	      Exponentiation and     Right
					      postfix Unicode
					superscripts
	6 (Highest)  expr[i]		      1-based array element  Left
					      indexing

   IMPLICIT MULTIPLICATION RULES
  Multiplication is inserted automatically at clear token boundaries:

     + Number followed by identifier, (, or √: 2x, 3π, 2(x + 1), 5√2.
     + Closing parenthesis ) followed by (, √, identifier, or number:
       (x + 1)(x - 1).
     + Variable or constant followed by ( or √: x(y + 1), π√2.	+
       Note: Adjacent letters like "ab" parse as a single two-character
       identifier, not a * b.

  BUILT-IN CONSTANTS

     + pi or π (U+03C0): π ≈ 3.141592653589793
     + τ (U+03C4): 2π ≈ 6.283185307179586
     + e: Euler's number ≈ 2.718281828459045

  BUILT-IN FUNCTIONS

     + 1-Argument Element-Wise: sin, cos, tan, asin, acos, atan, sinh,

PASSING ARRAYS
  Part 2: Passing Arrays To and From the Module in a Larger Code

  When integrating `M_unicode__eval` into a larger simulation or solver (such
  as a finite-element code, ODE integrator, or optimization loop), you do
  **not** want to recompile the expression string on every time step or
  iteration. Instead, follow a **3-phase lifecycle**:

  assing Arrays INTO the Module (`vars(:)%val`)

  •  Each variable in your expression corresponds to an index `i` in
     `var_names(1:N)` during `compile_expr_cp`.

  •  At runtime, you populate `vars(i)%val`—which is a standard Fortran
     `real(real64), allocatable :: val(:)` array.

  •  **For a 1D array variable** (e.g., a grid vector `u(1:nx)`): assign your
     Fortran array directly (`vars(1)%val = u`).

  •  **For a scalar variable** (e.g., time `t` or viscosity `nu`): wrap it in
     a 1-element array constructor (`vars(2)%val = [t]`).

  •  **For multi-dimensional arrays** (e.g., a 2D field `T(nx, ny)`): flatten
     it into a 1D vector with `reshape(T, [nx * ny])` when passing it in, then
     `reshape` the output back to `(nx, ny)`. Because all arithmetic is
     element-wise, flattening preserves exact element correspondence.

  2.  Receiving Arrays FROM the Module (`result(:)`)

  •  `eval_compiled(prog, result, ierr, vars)` takes `real(real64),
     allocatable, intent(out) :: result(:)`.

  •  Fortran automatically allocates `result` to match the broadcasted size:

     •	If the expression evaluates to a **vector** (e.g., `"nu * u +
	sin(x)"`), `size(result)` equals `size(u)`.

     •	If the expression reduces to a **scalar** (e.g., `"norm2(u) + t"`),
	`size(result)` equals `1` and the scalar value is `result(1)`.

  3.  Complete Integration Pattern (Solver Loop Example)

  ```fortran subroutine run_solver_step(x_grid, u_field, dt, t_now,
  formula_utf8, rhs_out, ierr) use, intrinsic :: iso_fortran_env, only: real64
  use M_unicode__eval implicit none

    real(real64), intent(in) :: x_grid(:), u_field(:) real(real64), intent(in)
    :: dt, t_now character(len=*), intent(in) :: formula_utf8 real(real64),
    intent(out) :: rhs_out(size(x_grid)) integer, intent(out) :: ierr

    ! Use 'save' (or store in a solver context struct) to compile only once
    type(compiled_expr), save :: compiled_rhs logical, save :: is_compiled =
    .false.

    type(cp_string) :: vnames(4) type(var_value) :: vars(4) real(real64),
    allocatable :: eval_out(:)

    ! PHASE 1: Compile once on first call if (.not. is_compiled) then
    vnames(1)%cp = utoc("x") vnames(2)%cp = utoc("u") vnames(3)%cp =
    utoc("dt") vnames(4)%cp = utoc("t")

      call compile_expr_cp(utoc(formula_utf8), compiled_rhs, ierr, vnames) if
      (ierr /= 0) return is_compiled = .true.  end if

    ! PHASE 2: Pack host Fortran arrays and scalars into var_value containers

    vars(1)%val = x_grid
      ! 1D array (size N)

    vars(2)%val = u_field
      ! 1D array (size N)

    vars(3)%val = [dt]
      ! Scalar wrapped as size-1 array

    vars(4)%val = [t_now]
      ! Scalar wrapped as size-1 array

    ! PHASE 3: Evaluate compiled bytecode and unpack result back into host
    array call eval_compiled(compiled_rhs, eval_out, ierr, vars) if (ierr /=
    0) return

    if (size(eval_out) == size(rhs_out)) then rhs_out = eval_out else if
    (size(eval_out) == 1) then ! Handle case where user's formula reduced to a
    constant/scalar (e.g., "0.0") rhs_out = eval_out(1) else ierr = 15 ! Shape
    mismatch with expected grid size end if end subroutine run_solver_step

  Performance Tip for Large Arrays (N > 100,000)

  In eval_compiled, every intermediate operation allocates a temporary array
  tmp(n) on the stack. If you evaluate expressions over massive grids
  (millions of points) inside a tight inner loop, you can avoid heap
  allocation churn by either:

      Chunking (Cache Blocking): Slice your large host arrays
      into L1/L2-cache-friendly blocks of 1024 or 4096 elements
      (x_grid(i:i+chunk-1)), evaluate each chunk with eval_compiled,
      and write directly into rhs_out(i:i+chunk-1).

      Thread-Safe OpenMP Parallelism: Because compiled_expr is read-only
      during eval_compiled and each thread maintains its own local vars
      and stack, you can safely parallelize across chunks using !$omp
      parallel do.

  val(:)) to var_value so large host arrays can be passed into eval_compiled
  without copying memory on every call?

  The M_unicode__regex(3) module loaded into the M_unicode(3) module compiles

  (`integer(int32)`). Because matching operates directly on Unicode codepoints

	use M_unicode, only : utoc,ctou

	m = regex_search(re, txt)

	if (m%matched) then
	   print g, 'Full match : ', ctou(txt(m%start_pos(0):m%end_pos(0)))
	   print g, 'Group 1	: ', ctou(txt(m%start_pos(1):m%end_pos(1)))
	   print g, 'Group 2	: ', ctou(txt(m%start_pos(2):m%end_pos(2)))
	else
	   print g, 'NO MATCH!'
	endif

	! Replace using backreferences
	new='$2 -> $1'
	print g, 'Old	     : ',ctou(txt)
	print g, 'New	     : ',new
	out = regex_replace(re, txt, utoc(new))
	print g, 'Output     : ', ctou(out)

	! Expected Output for Unicode-Aware Engines:
	print g, 'Expected   : ', 'Launch 🚀 -> Zürich now'

	! How it works:
	! \w+ matches the full word Zürich (including the ü) as Group 1,
	! and 🚀 is captured as Group 2.
	! The replacement swaps them around the arrow while leaving
	! "Launch " and " now" untouched.
     end program demo_regex_replace

  characters (LICENSE:MIT)

SYNOPSIS
  result = scan( string, set, [,back] )

     elemental integer(kind=KIND) function scan(string,set,back)

      type(unicode_type),intent(in) :: string

      type(unicode_type),intent(in) :: set
	 or
      character(len=*),intent(in)   :: set

      logical,intent(in),optional   :: back

CHARACTERISTICS
  •  STRING is a string of type unicode_type

  •  SET must be a string of type unicode_type or character

  •  BACK is a logical of default kind

  •  the result is an integer of default kind.

DESCRIPTION
  SCAN(3f) scans a STRING for any of the characters in a SET of characters.

  If BACK is either absent or equals .false., this function returns the
  position of the leftmost character of STRING that is in SET. If BACK equals
  .true., the rightmost position is returned. If no character of SET is found
  in STRING, the result is zero.

OPTIONS
  •  STRING : the string to be scanned

  •  SET : the set of characters which will be matched

  •  BACK : if .true. the position of the rightmost character matched is
     returned, instead of the leftmost.

RESULT
  If BACK is absent or is present with the value false and if STRING contains
  at least one character that is in SET, the value of the result is the
  position of the leftmost character of STRING that is in SET.

  If BACK is present with the value true and if STRING contains at least one
  character that is in SET, the value of the result is the position of the
  rightmost character of STRING that is in SET.

  The value of the result is zero if no character of STRING is in SET or if
  the length of STRING or SET is zero.

     use M_unicode, only : slurp, ut=>unicode_type
     use M_unicode, only : add_backslash, expand_backslash
     use M_unicode, only : assignment(=)
     implicit none
     type(ut),allocatable	  :: text(:)
     integer			  :: i
     character(len=:),allocatable :: iomsg

NAME
  SPLIT(3f) - [M_unicode:PARSE] parse a string into tokens, one at a time.
  (LICENSE:MIT)

SYNOPSIS
  call split (string, set, pos [, back])

     type(unicode_type),intent(in) :: string
     type(unicode_type),intent(in) :: set
     integer,intent(inout)	   :: pos
     logical,intent(in),optional   :: back

CHARACTERISTICS
  •  STRING is a scalar character variable

  •  SET is a scalar string variable

DESCRIPTION
  Find the extent of consecutive tokens in a string. given a string and a
  position to start looking for a token return the position of the end of the
  token. a set of separator characters may be specified as well as the
  direction of parsing.

  typically consecutive calls are used to parse a string into a set of tokens
  by stepping through the start and end positions of each token.

OPTIONS
  •  STRING : the string to search for tokens in.

  •  SET : Each character in set is a token delimiter. a sequence of zero or
     more characters in string delimited by any token delimiter, or the
     beginning or end of string, comprise a token. thus, two consecutive token
     delimiters in STRING, or a token delimiter in the first or last character
     of STRING, indicate a token with zero length.

  •  POS : on input, the position from which to start looking for the next
     separator from. This is typically the first character or the last
     returned value of POS if searching from left to right (ie. back is absent
     or .true.) or the last character or the last returned value of POS when
     searching from right to left (ie. when back is .FALSE.).

     If BACK is present with the value .TRUE., the value of pos shall be in
     the range 0 < POS <= len(STRING)+1; otherwise it shall be in the range 0
     <= POS <= len(STRING).

     So POS on input is typically an end of the string or the position of a
     separator, probably from a previous call to split but POS on input can be
     any position in the range 1 <= POS <= len(STRING). if POS points to a
     non-separator character in the string the call is still valid but it will
     start searching from the specified position and that will result
     (somewhat obviously) in the string from POS on input to the returned POS
     being a partial token.

  •  BACK : If BACK is absent or is present with the value .FALSE., POS is
     assigned the position of the leftmost token delimiter in string whose
     position is greater than POS, or if there is no such character, it is
     assigned a value one greater than the length of string. this identifies a
     token with starting position one greater than the value of POS on
     invocation, and ending position one less than the value of POS on return.

     If BACK is present with the value .TRUE., POS is assigned the position of
     the rightmost token delimiter in string whose position is less than POS,
     or if there is no such character, it is assigned the value zero. This
     identifies a token with ending position one less than the value of POS on
     invocation, and starting position one

     greater than the value of POS
       on return.

     program demo_split

     use M_unicode,	  only : unicode_type, assignment(=)
     use M_unicode,	  only : split, len, character
     use M_unicode,	  only : ut=>unicode_type
     implicit none
     character(len=*),parameter :: g='(*(g0,1x))'
     type(ut)			:: proverb
     type(ut)			:: delims
     type(ut),allocatable	:: array(:)
     integer			:: first
     integer			:: last
     integer			:: pos
     integer			:: i
	!
	delims= '=|; '
	!
	proverb="Más vale pájaro en mano, que ciento volando."
	call printwords(proverb)

	! there really are not spaces between these glyphs
	array=[ &
	 ut("七転び八起き。"), &
	 ut("転んでもまた立ち上がる。"), &
	 ut("くじけずに前を向いて歩いていこう。")]
	call printwords(array)
	!
	write(stdout,g)'OOP'
	array=proverb%split(ut(' '))
	write(stdout,'(*(:"[",a,"]"))')(character(array(i)),i=1,size(array))
     contains

     type(ut),intent(in) :: line
	pos = 0
	write(stdout,g)line%character(),len(line)
	do while (pos < len(line))
	    first = pos + 1
	    call split (line, delims, pos)
	    last = pos - 1
	    print g, line%character(first,last),first,last,pos
	end do
     end subroutine printwords

NAME
  SORT(3f) - [M_unicode:SORT] indexed hybrid quicksort of an array
  (LICENSE:MIT)

SYNOPSIS
  subroutine sort(data,index,comparison_func)

	  type(unicode_type),intent(in) 	  :: data(:)
	  integer,intent(out)			  :: indx(size(data))
	  procedure(unicode_comparator), optional :: comparison_func

DESCRIPTION
  A rank hybrid quicksort. The data is not moved. An integer array is
  generated instead with values that are indices to the sorted order of the
  data. This requires a second array the size of the input array, which for

  if another order is required.
    A custom collation function must take two values and return a logical
    value:

	 abstract interface
	    logical function unicode_comparator(str1, str2)
	    ! true if str1 comes before or is equal to str2 else false
	       import :: unicode_type
	       type(unicode_type), intent(in) :: str1, str2
	    end function unicode_comparator
	 end interface

BACKGROUND
  From Leonard J. Moss of SLAC:

  Here's a hybrid QuickSort I wrote a number of years ago. It's based on
  suggestions in Knuth, Volume 3, and performs much better than a pure
  QuickSort on short or partially ordered input arrays.

  This routine performs an in-memory sort of the first N elements of array
  DATA, returning into array INDEX the indices of elements of DATA arranged in
  ascending order. Thus,

	 DATA(INDX(1)) will be the smallest number in array DATA;
	 DATA(INDX(N)) will be the largest number in DATA.

  The original data is not physically rearranged. The original order of equal
  input values is not necessarily preserved.

  SORT(3f) uses a hybrid QuickSort algorithm, based on several suggestions in
  Knuth, Volume 3, Section 5.2.2. In particular, the "pivot key" [my term] for
  dividing each subsequence is chosen to be the median of the first, last, and
  middle values of the subsequence; and the QuickSort is cut off when a
  subsequence has 9 or fewer elements, and a straight insertion sort of the
  entire array is done at the end. The result is comparable to a pure
  insertion sort for very short arrays, and very fast for very large arrays
  (of order 12 micro-sec/element on the 3081K for arrays of 10K elements). It
  is also not subject to the poor performance of the pure QuickSort on
  partially ordered data.

  Complex values are sorted by the magnitude of sqrt(r**2+i**2).

  •  Created: sortrx(3f): 15 Jul 1986, Len Moss

  •  saved from url=(0044)http://www.fortran.com/fortran/quick_sort2.f

  •  changed to update syntax from F77 style; John S. Urban 20161021

  •  generalized from only real values to include other intrinsic types; John
     S. Urban 20210110

  •  type(unicode_type) version JSU 2025-09-20. See M_sort for other types.

	function case_insensitive_compare(str1, str2) result (answer)
	! case-insensitive comparison
	type(unicode_type), intent(in) :: str1, str2
	logical 		       :: answer
	   answer = lower(str1) < lower(str2)
	end function case_insensitive_compare
     !
     ! German dictionaries follow specific guidelines
     ! for sorting umlauts (a, o, u) and the eszett (β).
     !
     ! Standard Dictionary Rules (DIN 5007-1, Variant 1)
     !
     ! In traditional German dictionaries, words with umlauts are treated as
     ! their base vowels and the letter β is treated as ss. That is ...
     !
     !	   Amlausts:
     !	       ä is sorted as a
     !	       ö is sorted as o
     !	       ü is sorted as u
     !	   Eszett:
     !	       β is sorted as ss
     !
     !	along with a similar exchange for their uppercase cases.
     !
     !	If normalized words are identical, the umlaut comes second.
     !
     !	The multibyte characters with an umlaut may be a single Unicode
     !	codepoint or the associated Latin character followed by the
     !	combining dieresis modifier U+0308.
     !
     ! Note for reference that depending on the context (such as phone books
     ! or digital systems), other official standards apply:
     !
     !	 + Phone Book Rules (DIN 5007-1, Variant 2): Umlauts are expanded by
     !	   adding an e after the vowel (ä = ae, ö = oe, ü = ue). As in Variant
     !	   1 the letter β is treated as ss.
     !
     !	 + Strict / Character-by-Character Order: Some systems or older indexes
     !	   place umlauts right after their base letter as separate characters
     !	   (a, a, b, c...).
     !
     function german_dictionary(lhs, rhs) result(is_lle)
     !
     ! @(#) Lexically compare two strings using German dictionary order
     !
     type(ut), intent(in) :: lhs, rhs
     logical		  :: is_lle

	is_lle = nmi(lhs) < nmi(rhs)

     end function german_dictionary

     function nmi(in) result (out)
     type(ut),intent(in)  :: in
     type(ut)		  :: out
     integer,allocatable  :: codes(:)
     integer		  :: i, ikeep
     integer,parameter	  :: comb=2*int(z'308'), &
      &  um_lower_a    =  2*int(z'E4'),       &
      &  um_lower_o    =  2*int(z'F6'),       &
      &  um_lower_u    =  2*int(z'FC'),       &
      &  um_upper_a    =  2*int(z'C4'),       &
      &  um_upper_o    =  2*int(z'D6'),       &
      &  um_upper_u    =  2*int(z'DC'),       &
      &  lower_a       =  2*ichar('a')+1,     &
      &  lower_o       =  2*ichar('o')+1,     &
      &  lower_u       =  2*ichar('u')+1,     &
      &  upper_a       =  2*ichar('A')+1,     &
      &  upper_o       =  2*ichar('O')+1,     &
      &  upper_u       =  2*ichar('U')+1,     &
      &  lower_eszett  =  2*int(z'DF'),       &
      &  upper_eszett  =  2*int(z'1E9E'),     &
      &  lower_s       =  2*int(ichar('s')),  &
      &  upper_s       =  2*int(ichar('S'))

	! if every letter is a beta and first letter is COMB
	! (which should not happen) longest index required would be
	allocate(codes(0:2*len(in)))

	ikeep=0
	do concurrent ( i=1:len(in) )
	   if(in%codes(i) == comb)then
	      codes(ikeep)=codes(ikeep)+1
	   else
	      ikeep=ikeep+1
	      codes(ikeep)=in%codes(i)*2
	      select case(codes(ikeep))
	       case(:127*2)
	       case(um_lower_a); codes(ikeep)=lower_a
	       case(um_lower_o); codes(ikeep)=lower_o
	       case(um_lower_u); codes(ikeep)=lower_u
	       case(um_upper_a); codes(ikeep)=upper_a
	       case(um_upper_o); codes(ikeep)=upper_o
	       case(um_upper_u); codes(ikeep)=upper_u
	       case(lower_eszett)
		  codes(ikeep:ikeep+1)=lower_s
		  ikeep=ikeep+1
	       case(upper_eszett)
		  codes(ikeep:ikeep+1)=upper_s
		  ikeep=ikeep+1
	      end select
	   endif
	enddo
	! codes are no longer valid Unicode codepoints for the input but
	! can be used to compare lexically to another normalized value
	! only because needed some method of telling if an A,O,U used to
	! have an umlaut
	out=codes(1:ikeep)

     end function nmi

     end program demo_sort

NAME

SYNOPSIS
  result = adjustr(string,glyphs)

      type(unicode_type)	    :: adjustr
      type(unicode_type),intent(in) :: string
      integer,intent(in),optional   :: glyphs

CHARACTERISTICS
  •  STRING is a string variable

  •  GLYPHS is a default integer

  •  the return value is a string variable

DESCRIPTION
  ADJUSTR(3f) right-justifies a string by removing trailing spaces. Spaces are
  inserted at the start of the string as needed to retain the original length
  unless an explicit return length is specified by the GLYPHS parameter.

OPTIONS
  •  STRING : the string to right-justify

RESULT

  at the start of string.

NAME
  TOKENIZE(3f) - [M_unicode:PARSE] Parse a string into tokens.	(LICENSE:MIT)

SYNOPSIS
  TOKEN form (returns array of strings)

     subroutine tokenize(string, set, tokens [, separator])

      type(unicode_type),intent(in) :: string
      type(unicode_type),intent(in) :: set
      type(unicode_type),allocatable,intent(out) :: tokens(:)
      type(unicode_type),allocatable,intent(out),optional :: separator(:)

  ARRAY BOUNDS form (returns arrays defining token positions)

     subroutine tokenize (string, set, first, last)

      type(unicode_type),intent(in) :: string
      type(unicode_type),intent(in) :: set
      integer,allocatable,intent(out) :: first(:)
      integer,allocatable,intent(out) :: last(:)

CHARACTERISTICS
  •  STRING ‐ a scalar of type string. It is an INTENT(IN) argument.

  •  SET ‐ a scalar of type string with the same kind type parameter as
     STRING. It is an INTENT(IN) argument.

  •  SEPARATOR ‐ (optional) shall be of type string. It is an
     INTENT(OUT)argument. It shall not be a coarray or a coindexed object.

  •  TOKENS ‐ of type string. It is an INTENT(OUT) argument. It shall not be a
     coarray or a coindexed object.

  •  FIRST,LAST ‐ an allocatable array of type integer and rank one. It is an
     INTENT(OUT) argument. It shall not be a coarray or a coindexed object.

DESCRIPTION
  TOKENIZE(3f) parses a string into tokens. There are two forms of the
  subroutine TOKENIZE(3f).

  •  The token form returns an array with one token per element, all of the
     same length as the longest token.

  •  The array bounds form returns two integer arrays. One contains the
     beginning position of the tokens and the other the end positions.

  Since the token form pads all the tokens to the same length the original
  number of trailing spaces of each token accept for the longest is lost.

  The array bounds form retains information regarding the exact token length
  even when padded by spaces.

OPTIONS
  • STRING : The string to parse into tokens.

  •  SET : Each character in SET is a token delimiter. A sequence of zero or
     more characters in STRING delimited by any token delimiter, or the
     beginning or end of STRING, comprise a token. Thus, two consecutive token
     delimiters in STRING, or a token delimiter in the first or last character
     of STRING, indicate a token with zero length.

  •  TOKENS : It shall be an allocatable array of rank one with deferred
     length. It is allocated with the lower bound equal to one and the upper
     bound equal to the number of tokens in STRING, and with character length
     equal to the length of the longest token.

     The tokens in STRING are assigned in the order found, as if by intrinsic
     assignment, to the elements of TOKENS, in array element order.

  •  FIRST : shall be an allocatable array of type integer and rank one.  It
     is an INTENT(OUT) argument. It shall not be a coarray or a coindexed
     object.

     It is allocated with the lower bound equal to one and the upper bound
     equal to the number of tokens in STRING. Each element is assigned, in
     array element order, the starting position of each token in STRING, in
     the order found.

     If a token has zero length, the starting position is equal to one if the
     token is at the beginning of STRING, and one greater than the position of
     the preceding delimiter otherwise.

  •  LAST : It is allocated with the lower bound equal to one and the upper
     bound equal to the number of tokens in STRING. Each element is assigned,
     in array element order, the ending position of each token in STRING, in
     the order found.

     If a token has zero length, the ending position is one less than the
     starting position.

  Sample of uses

       !

NAME
  UPPER(3f) - [M_unicode:CASE] changes a string to uppercase (LICENSE:MIT)

SYNOPSIS
  pure elemental function upper(str) result (string)

       type(unicode_type),intent(in) :: str
       type(unicode_type)	     :: string

DESCRIPTION
  upper(string) returns a copy of the input string with all characters
  converted to uppercase, assuming Unicode character sets are being used.

OPTIONS
  str
    string to convert to uppercase

RETURNS
  upper
    copy of the input string with all characters converted to uppercase.

TRIVIA
  The terms "uppercase" and "lowercase" date back to the early days of the
  mechanical printing press. Individual metal alloy casts of each needed
  letter, or punctuation symbol, were meticulously added to a press block, by
  hand, before rolling out copies of a page. These metal casts were stored and
  organized in wooden cases. The more often needed miniscule letters were
  placed closer to hand, in the lower cases of the work bench. The less often
  needed, capitalized, majuscule letters, ended up in the harder to reach
  upper cases.

EXAMPLES
  Sample program:

     program demo_upper
     use iso_fortran_env, only : stdout => output_unit
     use M_unicode,	  only : upper, unicode_type, assignment(=)
     use M_unicode,	  only : ut => unicode_type, operator(==)
     implicit none
     character(len=*),parameter :: g='(*(g0))'
     type(unicode_type) 	:: pangram
     type(unicode_type) 	:: diacritics
     type(unicode_type) 	:: expected
	!
	! a sentence containing every letter of the English alphabet
	! often used to test telegraphs since the advent of the 19th century
	! and as an exercise repetitively generated in typing classes
	pangram  = "The quick brown fox jumps over the lazy dog."
	expected = "THE QUICK BROWN FOX JUMPS OVER THE LAZY DOG."
	call test(pangram,expected)
	!
	! Slovak pangram
	pangram    = 'Vypätá dcéra grófa Maxwella s IQ nižším ako &
	&kôň núti čeľaď hrýzť hŕbu jabĺk.'
	expected   = 'VYPÄTÁ DCÉRA GRÓFA MAXWELLA S IQ NIŽŠÍM AKO &
	&KÔŇ NÚTI ČEĽAĎ HRÝZŤ HŔBU JABĹK.'
	call test(pangram,expected)
	!
	! contains each special Czech letter with diacritics exactly once
	print g,'("A horse that was too yellow-ish moaned devilish odes")'
	diacritics = 'Příliš žluťoučký kůň úpěl ďábelské ódy.'
	expected   = 'PŘÍLIŠ ŽLUŤOUČKÝ KŮŇ ÚPĚL ĎÁBELSKÉ ÓDY.'
	call test(diacritics,expected)
     contains
     subroutine test(in,expected)
     type(unicode_type),intent(in) :: in
     type(unicode_type),intent(in) :: expected
     type(unicode_type) 	   :: uppercase
     character(len=*),parameter    :: nl=new_line('A')
	write(stdout,g)in%character()
	uppercase=upper(in)
	write(stdout,g)uppercase%character()
	write(stdout,g)merge('PASSED','FAILED',uppercase == expected ),nl

NAME
  UTF8_TO_CODEPOINTS(3f) - [M_unicode:CONVERSION] Convert UTF-8-encoded data
  to Unicode codepoints (LICENSE:MIT)

SYNOPSIS
  pure subroutine utf8_to_codepoints(utf8,codepoints,nerr)

      character(len=1),intent(in)     :: utf8(:)
      !  or
      character(len=*),intent(in)     :: utf8
      !
      integer,allocatable,intent(out) :: codepoints(:)
      integer,intent(out)	      :: nerr

CHARACTERISTICS
  •  UTF8 is a scalar CHARACTER variable or array of single-byte CHARACTER
     values

  •  the returned values in CODEPOINTS are of default INTEGER kind

  •  the error flag NERR is default integer kind

DESCRIPTION
  UTF8_TO_CODEPOINTS(3f) takes either a scalar CHARACTER variable or an array
  of CHARACTER(LEN=1) bytes which are treated as a stream of bytes
  representing UTF-8-encoded data and converted to an INTEGER array containing
  Unicode codepoint values for each glyph.

  •  UTF8 : Scalar CHARACTER string or single-character array of CHARACTER
     variables assumed to represent a stream of bytes containing data encoded
     as UTF-8 text.

  •  CODEPOINTS : An INTEGER array of Unicode codepoint values representing
     the glyphs found in STRING

  •  NERR : Zero if no error occurred. If not zero the stream of bytes could
     not be completely converted to UTF-8 characters.

  Sample program

     program demo_utf8_to_codepoints
     use m_unicode, only : utf8_to_codepoints
     implicit none
     character(len=*),parameter   :: string ='Noho me ka hau’oli' !(Be happy)
     character(len=1),allocatable :: bytes(:)
     character(len=*),parameter   :: solid='(*(g0))'
     character(len=*),parameter   :: space='(*(g0,1x))'
     character(len=*),parameter   :: z='(a,*(z0,1x))'
     integer,allocatable	  :: codepoints(:)
     integer			  :: nerr
     integer			  :: i
     ! BASIC USAGE: SCALAR CHARACTER VARIABLE
       write(*,solid)'STRING:',string
       call utf8_to_codepoints(string,codepoints,nerr)
       write(*,space)'CODEPOINTS:', codepoints
       write(*,z)'HEXADECIMAL CODEPOINTS:', codepoints
     !
       write(*,space)'How long is this string in glyphs? '
       write(*,space)size(codepoints)
       write(*,space)'How long is this string in bytes? '
       write(*,space)len(string)
     !
     ! BASIC USAGE: ARRAY OF BYTES
       bytes=[(string(i:i),i=1,len(string))]
       write(*,solid)'STRING:',bytes
       call utf8_to_codepoints(bytes,codepoints,nerr)
       write(*,space)'CODEPOINTS:', codepoints
       write(*,z)'HEXADECIMAL CODEPOINTS:', codepoints
     !
       write(*,space)'How long is this string in glyphs? '
       write(*,space)size(codepoints)
       write(*,space)'How long is this string in bytes? '
       write(*,space)size(bytes)

     end program demo_utf8_to_codepoints

NAME
  expand_env(3f) - [M_unicode:QUERY] implements basic shell-style variable
  expansion (LICENSE:PD)

SYNOPSIS
  function expand_env(in) result(out)

      character(len=1),intent(in)  :: in
      character(len=:),allocatable :: out

      !  or

      type(unicode_type),intent(in)  :: in
      type(unicode_type),allocatable :: out

DESCRIPTION
  expand_env(3f) implements basic shell-style variable expansion for

  environment variables and some pre-defined names.
    That is, it scans for or pre-defined substitutions.

  Given a string containing "$VAR" and "${VAR}" strings it replaces it

  with the value of the environment variable VAR .
    If the environment variable name returns a null string or the variable is
    not defined, the following predefined names will be expanded:

    •  FILE: same as value returned by GET_COMMAND_ARGUMENT(0,string)

    •  0,1,2,....: same as value returned by GET_COMMAND_ARGUMENT(N,string)

    •  *: all command line arguments from 1 to N joined with a space separator

    •  COMPILER: same as value returned by COMPILER_VERSION() intrinsic

    •  COMPILER_OPTIONS: same as value returned by COMPILER__OPTIONS()
       intrinsic

    •  DATE: returns a string of the form CCYY-MM-DD

    •  TIME: returns a string of the form HH:MM:SS

    •  ZONE: returns a string of the form (+-)HHMM, representing the
       difference with respect to Coordinated Universal Time (UTC)

    •  TIMESTAMP: CCYY-MM-DDTHH:MM:SS

OPTIONS
  A string containing names of the syntax $VAR or ${VAR} in addition to plain
  text not to be expanded.

RETURNS
  A string where $VAR and ${VAR} are replaced with the value of the
  environment variable. If the returned value is null several predefined
  variable names will be expanded per the table above.

NAME
  issensitive(3f) - [M_unicode:CASE] determine if character is case-sensitive
  (LICENSE:MIT)

SYNOPSIS
  pure elemental function issensitive(str) result (mask)

       type(unicode_type),intent(in) :: str
       logical			     :: mask

DESCRIPTION
  issensitive(string) returns true if the character is considered to be case-
  sensitive, else it returns false.

OPTIONS
  str
    string to query, only first character is tested.

RETURNS
  mask
    returns true if character is considered case-sensitive; otherwise is
    returns false.

EXAMPLES
  Sample program:

     program demo_issensitive
     use iso_fortran_env, only : stdout => output_unit
     use M_unicode,	  only : issensitive, len, assignment(=)
     use M_unicode,	  only : ut => unicode_type, lower, upper
     use M_unicode,	  only : ch => character
     implicit none
     type(ut)			:: lines(4), line, low, upp, diff
     character(len=*),parameter :: g = '(*(g0))'
     integer			:: i
     integer			:: j

	lines(1) = 'P{äg15🍐 🌦6^c0ßßẞ)🌛🌢;4279OTRNiü!zx🌴|'
	lines(2) = 'ＡＢＣＤＥＦＧＨＩＪＫＬＭＮＯＰＱＲＳＴＵＶＷＸＹＺ'
	lines(3) = 'abcdefghijklmnopqrstuvwxyz ABCDEFGHIJKLMNOPQRSTUVWXYZ'
	lines(4) = 'v:Hk~dqyUBm🌟🌬lEwöJ🌀A,f[]8🌷(S}🍍3'

	do j=1,size(lines)
	   ! given a line
	   line=lines(j)
	   ! generate a lowercase and uppercase copy
	   low=lower(line)
	   upp=upper(line)

	   ! determining which characters are different in the
	   ! lowercase and uppercase versions lets us determine
	   ! which characters are case-sensitive, at least as
	   ! far as the upper(3f) and lower(3f) functions
	   print g, 'ORIGINAL	   :',line%ch()
	   print g, 'LOWERCASE COPY:',low%ch()
	   print g, 'UPPERCASE COPY:',upp%ch()
	   diff=pack(line%codes,low%codes == upp%codes)
	   print g, 'NO CASE CHANGE:',diff%ch()
	   diff=pack(line%codes,low%codes /= upp%codes)
	   print g, 'CASE CHANGED  :',diff%ch()

	   ! the same characters should be identified by the
	   ! issensitive(3f) function
	   write(*,g,advance='no')'SENSITIVE	 :'
	   do i = 1,len(line)
	      if (issensitive(line%sub(i,i))) then
		 write(*,g,advance='no')line%ch(i,i)
	      endif
	   enddo
	   write(*,g)

	   write(*,g,advance='no')'NOT SENSITIVE :'
	   do i = 1,len(line)
	      if (.not.issensitive(line%sub(i,i))) then
		 write(*,g,advance='no')line%ch(i,i)
	      endif
	   enddo
	   write(*,g)
	   write(*,g)
	enddo

     end program demo_issensitive

NAME
  INDEX(3f) - [M_unicode:SEARCH] Position of a substring within a string
  (LICENSE:MIT)

SYNOPSIS
  result = index( string, substring [,back] [,kind] )

     elemental integer(kind=KIND) function index(string,substring,back,kind)

      character(len=*,kind=KIND),intent(in) :: string
      character(len=*,kind=KIND),intent(in) :: substring
      logical(kind=**),intent(in),optional :: back
      integer(kind=**),intent(in),optional :: kind

CHARACTERISTICS
  •  STRING is a character variable of any kind

  •  SUBSTRING is a character variable of the same kind as STRING

  •  BACK is a logical variable of any supported kind

  •  KIND is a scalar integer constant expression.

DESCRIPTION
  INDEX(3f) returns the position of the start of the leftmost or rightmost
  occurrence of string SUBSTRING in STRING, counting from one. If SUBSTRING is
  not present in STRING, zero is returned.

OPTIONS
  •  STRING : string to be searched for a match

  •  SUBSTRING : string to attempt to locate in STRING

  •  BACK : If the BACK argument is present and true, the return value is the
     start of the rightmost occurrence rather than the leftmost.

  •  KIND : if KIND is present, the kind type parameter is that specified by
     the value of KIND; otherwise the kind type parameter is that of default
     integer type.

RESULT
  The result is the starting position of the first substring SUBSTRING found
  in STRING.

  If the length of SUBSTRING is longer than STRING the result is zero.

  If the substring is not found the result is zero.

  If BACK is .true. the greatest starting position is returned (that is, the
  position of the right‐most match). Otherwise, the smallest position starting
  a match (ie. the left‐most match) is returned.

  The position returned is measured from the left with the first character of
  STRING being position one.

  Otherwise, if no match is found zero is returned.

NAME
  keyword_expand(3f) - [M_unicode] substitute HTML-like keywords with other
  strings (LICENSE:MIT)

SYNOPSIS
  function keyword_expand(string) result (expanded)

	 ! scalar
	 character(len=*),intent(in) :: string
	 ! or
	 type(unicode_type),intent(in) :: string

	 ! or array

	 ! or

NAME
  M_unicode__regex(3f) - [M_unicode:REGULAR EXPRESSIONS] Overview of UTF-8
  codepoint Regular Expression engine (LICENSE:MIT)

SYNOPSIS
  Syntax

      use M_unicode__regex, only: regex_t, match_t, &
			   regex_compile, regex_search, regex_replace, &
      use M_unicode__convert, only: utoc, ctou

      type(regex_t) :: re
      type(match_t) :: m

   CODEPOINT
  integer(int32), allocatable
    :: pat(:), txt(:), repl(:), out(:) character(len=:), allocatable :: str

       pat  = utoc(str)
       re   = regex_compile(pat)
       m    = regex_search(re, txt [, start_offset])
       out  = regex_replace(re, txt, repl [, replace_all])
       str  = ctou(out)

   STRING
  type(unicode_type) :: pat, txt, repl, out

       re   = regex_compile(pat)
       m    = regex_search(re, txt [, start_offset])
       out  = regex_replace(re, txt, repl [, replace_all])

   CHARACTER
  character(len=:),allocatable :: pat, txt, repl, out

       re   = regex_compile(pat)
       m    = regex_search(re, txt [, start_offset])
       out  = regex_replace(re, txt, repl [, replace_all])

DESCRIPTION
  The M_unicode__regex(3f)
    module implements a self-contained, Unicode-aware regular expression
    compiler and backtracking Nondeterministic Finite Automaton (NFA) virtual
    machine in standard Fortran 2008.

  The procedures are generic and operator on three different types --
  TYPE(UNICODE_TYPE) variables, CHARACTER variables or CODEPOINTS(integer
  arrays composed of valid Unicode codepoint values).

  For the CODEPOINT mode all pattern compilation, text matching, and

  string replacement operate directly on arrays of UTF-8/UTF-32
    scalar codepoints (integer(int32)).

  Helper functions utoc(3) and ctou(3) convert between standard UTF-8 encoded
  Fortran character strings and codepoint arrays.

  For the CHARACTER mode the same entities are all CHARACTER variables.

  For the STRING mode all the same entities are all TYPE(UNICODE_TYPE).

   FEATURES
  •  **Native UTF-32 Matching & UTF-8 Codecs:** Built-in `utf8_to_codepoints`
     and `codepoints_to_utf8` converters with `U+FFFD` replacement handling
     for malformed byte sequences.

  •  **Submatch Captures:** Supports full-match tracking (`group 0`) plus up
     to 9 parenthesized capture groups (`1..9`).

  •  **Backreference Replacement:** Global or single-match string replacement
     via `regex_replace` supporting `$0..$9` and `\0..\9` backreferences.

  •  **Greedy & Non-Greedy Quantifiers:** Full support for `*`, `+`, `?`,
     `{n}`, `{n,}`, `{n,m}` and their lazy counterparts `*?`, `+?`, `??`,
     `{n,}?`, `{n,m}?`.

  •  **Character Classes & Shorthands:** Bracket expressions `[...]`, negated
     classes `[^...]`, ranges `[a-z]`, and Unicode-aware shorthands (`\d`,
     `\D`, `\w`, `\W`, `\s`, `\S`).

  •  **Anchors & Word Boundaries:** Multiline start/end anchors (`^`, `$`) and
     Unicode-aware word boundaries (`\b`, `\B`).

  •  **Zero External Dependencies:** Pure Fortran 2008.

PUBLIC DERIVED TYPES
  type :: regex_t
    Compiled regular expression bytecode container.

	     Components:

		 prog(:)     Allocatable array of VM instructions (private layout).
		 num_groups  Integer count of capture groups including group 0
			     (1 <= num_groups <= 10).
		 valid	     Logical flag set to .true. if compilation succeeded,
			     or .false. if the pattern contained a syntax error
			     or exceeded group limits.

  type :: match_t
    Result structure returned by regex_search(3).

	     Components:

		 matched	 Logical flag; .true. if a match was found.
		 start_pos(0:9)  starting codepoint index for each capture
				 group. Group 0 is the entire match; groups 1..9
				 are parenthesized subexpressions.
		 end_pos(0:9)	 1-based inclusive ending codepoint index for each
				 capture group. For an unmatched group or a zero-
				 length match, end_pos(i) < start_pos(i).

PUBLIC PROCEDURES
  regex_compile(pattern) result(re)

	     Compiles a UTF-32 regular expression pattern into NFA bytecode.

	     Arguments:

		 pattern     integer(int32), intent(in) :: pattern(:)
			     Array of Unicode codepoints representing the regex.

	     Return Value:
		 re	     type(regex_t)
			     Inspect re%valid before passing re to regex_search
			     or regex_replace. Invalid patterns return
			     re%valid = .false. without aborting.

  regex_search(re, text [, start_offset]) result(m)

	     Searches text for the leftmost match of the compiled expression re.

	     Arguments:
		 re	       type(regex_t), intent(in)
			       Compiled regular expression. If .not. re%valid,
			       returns m%matched = .false. immediately.
		 text	       integer(int32), intent(in) :: text(:)
			       Subject text as an array of UTF-32 codepoints.
		 start_offset  integer, intent(in), optional
			       1-based codepoint index at which to begin searching
			       (defaults to 1; values < 1 are clamped to 1).

	     Return Value:
		 m	       type(match_t)
			       If m%matched is .true., the matched slice for group
			       g (0 <= g < re%num_groups) is:
				   text(m%start_pos(g) : m%end_pos(g))
			       whenever m%start_pos(g) >= 1 and
			       m%end_pos(g) >= m%start_pos(g).

  regex_replace(re, text, replacement [, replace_all]) result(out_text)

	     Replaces matches of re in text with replacement, expanding capture
	     group backreferences.

	     Arguments:

		 re	      type(regex_t), intent(in)
			      Compiled regular expression.
		 text	      integer(int32), intent(in) :: text(:)
			      Subject text as an array of UTF-32 codepoints.
		 replacement  integer(int32), intent(in) :: replacement(:)
			      Replacement template as UTF-32 codepoints. Supports
			      backreferences $0..$9 and \0..\9, where group 0 is
			      the full match and 1..9 are capture groups. Literal
			      '$' or '\' may be escaped by doubling ('$$' or '\\').
		 replace_all  logical, intent(in), optional
			      If .true. (the default), replaces all non-overlapping
			      matches in text. If .false., replaces only the first
			      match.

	     Return Value:

		 out_text     integer(int32), allocatable :: out_text(:)
			      Newly allocated UTF-32 codepoint array containing
			      the resulting text.

REGULAR EXPRESSION SYNTAX
  Atoms

	     c		 Matches literal Unicode codepoint c.
	     .		 Matches any single codepoint except Line Feed (U+000A).
	     [abc]	 Character class matching any listed codepoint or range
			 (e.g., [a-z0-9_]). Supports \d, \w, \s, \n, \r, \t.
	     [^abc]	 Negated character class matching any codepoint not in the
			 listed set (including Line Feed).
	     (expr)	 Capturing group (up to 9 groups, numbered 1..9 left-to-
			 right by opening parenthesis).
	     \x 	 Escaped character or shorthand class (see below).

  Shorthand Classes & Escapes

	     \d, \D	 ASCII digit [0-9] (\d) or non-digit (\D).
	     \w, \W	 Word character (\w): [0-9A-Za-z_] plus Latin-1 and Latin
			 Extended letters (U+00C0..U+00D6, U+00D8..U+00F6,
			 U+00F8..U+024F). \W matches any non-word codepoint.
	     \s, \S	 Whitespace (\s): ASCII space (U+0020), \t..\r (U+0009..
			 U+000D), non-breaking space (U+00A0), and Unicode spaces
			 (U+2000..U+200A). \S matches any non-whitespace codepoint.
	     \n, \r, \t  Line Feed (U+000A), Carriage Return (U+000D), Tab (U+0009).
	     \\, \., ... Escapes any metacharacter to match it literally.

  Anchors & Zero-Width Assertions

	     ^		 Matches start of text or immediately after Line Feed (U+000A).
	     $		 Matches end of text or immediately before Line Feed (U+000A).
	     \b 	 Matches a Unicode word boundary (transition between \w and
			 \W, including start/end of text).
	     \B 	 Matches a position that is not a word boundary.

  Quantifiers (Greedy and Non-Greedy)

    •  0 or more times (greedy).

    •  1 or more times (greedy).

    ?  0 or 1 time (greedy).

    {n}
       Exactly n times (0 <= n <= 1000).

    {n,}
       n or more times (greedy).

    {n,m}
       Between n and m times inclusive (greedy, 0 <= n <= m <= 1000).  *?, +?,
       ??, {n,}?, {n,m}?  Non-greedy (lazy) variants matching as few
       repetitions as possible while still allowing the overall pattern to
       succeed.

  Alternation & Precedence

	     expr1|expr2 Matches expr1 or, if that fails, expr2.
	     Precedence from highest to lowest:
		 1. Atoms, character classes, escapes, parenthesized groups
		 2. Quantifiers (*, +, ?, {n,m})
		 3. Concatenation
		 4. Alternation (|)

  LIMITATIONS & NOTES

  •  Maximum capture groups is 9 (MAX_GROUPS = 10 including group 0).

     •	Maximum repetition bound in {n,m} is 1000.

     •	The VM uses recursive backtracking; pathological nested quantifiers
	such as (a+)+b on non-matching input can exhibit exponential time and
	stack depth.

EXAMPLE
  Sample program

     program demo_M_unicode__regex
     use iso_fortran_env, only: int32
     use M_unicode__regex, only : regex_t, match_t
     use M_unicode__regex, only : regex_compile, regex_search, regex_replace
     use M_unicode__convert, only : utoc, ctou
     implicit none
     type(regex_t)		  :: re
     type(match_t)		  :: m
     integer(int32), allocatable  :: txt(:), out(:)
     character(len=:),allocatable :: pat
	pat='(\w+)\s+(🚀|🛸)'
	re = regex_compile(utoc(pat))
	if (.not. re%valid) error stop 'Invalid pattern'
	txt = utoc('Mission Zürich 🚀 ready')
	m = regex_search(re, txt)
	if (m%matched) then
	   print '(A)', 'City:	' // &
	   & ctou(txt(m%start_pos(1):m%end_pos(1)))
	   print '(A)', 'Craft: ' // &
	   & ctou(txt(m%start_pos(2):m%end_pos(2)))
	endif
	out = regex_replace(re, txt, utoc('$2 -> $1'))
	print '(A)', 'Swapped: ' // ctou(out)
     end program demo_M_unicode__regex

NAME
  str_to_real(3f) ‐ [M_unicode:TYPE] subroutine returns numeric value from
  string (LICENSE:MIT)

SYNOPSIS
  pure elemental subroutine str_to_real(str, val, ierr)

	  character(len=*), intent(in)	:: str
	  real(real64),     intent(out) :: val
	  integer,	    intent(out) :: ierr

DESCRIPTION
  STR_TO_REAL(3f) returns a numeric value given a string representing a
  numeric value; without using internal I/O (the conventional method in
  Fortran).

  The procedure utilitizes full IEEE 754 support for literal Inf/Infinity, NaN
  (case-insensitive), and signed -0.0 preservation, while keeping the
  procedure pure elemental.

  STR_TO_REAL(3f) works with any G‐format input, This includes whole number,
  float, and exponential syntax. That is, input strings may contain signs
  (+/-), decimal points, and scientific notation (e/E/d/D).

  The output is a double-precision real (real(real64)), along with an integer
  error flag (ierr).

OPTIONS
  chars
    input string to read numeric value from

RETURNS
  valu
    DOUBLEPRECISION numeric value returned.

  ierr
    error flag (0 == no error)

   KEY FEATURES OF THIS IMPLEMENTATION
  •  ASCII Digit Conversion. Uses iachar(c) - iachar('0') rather than
     hardcoded ASCII offsets like 48 to stay portable across compiler
     environments.

  •  No Internal I/O Overhead. Avoids Fortran's internal READ(3f) statement,
     making it faster in tight loops and safe to use inside pure or elemental
     procedures or inside parallel OpenMP/GPU regions where I/O is restricted.

  •  Error Handling (ierr). Safely catches empty strings (1), misplaced
     decimals (2), malformed exponents (3 or 5), and stray non-numeric
     characters (4) without crashing the program.

  •  The subroutine is pure (and elemental) so it can process arrays directly,
     To maintain this feature the first and last non-space character indices
     are located directly on the input, avoiding dynamic heap allocation. The
     intent is to keeps the procedure completely side-effect-free and
     reasonably fast while providing input valication.

     Pure procedures are required for optimal performance or even appearance
     iin do concurrent loops, OpenMP simd/target regions, or directly on
     multi-dimensional string arrays.

  •  Zero Heap Allocations. This avoids allocating memory inside the procedure
     and keeps the CPU cache footprint minimal.

  •  overflow protection uses Fortran intrinsic inquiry functions (huge(),
     maxexponent(), and minexponent()) rather than hardcoded magic numbers.

     overflow is guarded agains in in three places:

       1.  The Exponent Integer (exp_val): Prevents integer overflow if a
	   malicious or malformed string has a massive exponent like
	   1e99999999999.

       2.  The Mantissa (int_part): This prevents floating-point overflow
	   before the exponent is even applied (e.g., a 400-digit string with
	   no e marker).

       e. The Final Scaled Result (val):
	   Checks the combined order of magnitude against maxexponent(val)
	   before raising 10.0 to a power that would produce Infinity or
	   trigger a floating-point exception.

   SUMMARY OF PROTECTIONS ADDED
  Error Code 6 (Numeric Overflow): Returned cleanly as Infinity without
  halting execution if the value exceeds huge(1.0_real64) (~1.79769e308).

  Exact Edge-Case Checking at 10^308: Distinguishes between valid numbers near
  the ceiling (such as "1.79e308") and overflowing numbers at the same order
  of magnitude (such as "1.80e308").

  Subnormal / Underflow Handling: Numbers smaller than ~10^-324 underflow
  cleanly to 0.0_real64. Numbers in the subnormal range (10^-308 to 10^-324)
  are scaled in two steps (1.0e-300 first) so 10.0_real64 ** exp does not
  prematurely evaluate to zero before multiplying by the mantissa.

  Zero-Mantissa Safety: Inputs like "0.0e5000" evaluate to 0.0_real64 with
  ierr = 0 rather than falsely triggering an overflow.

  Case-Insensitive Inf, Infinity, and NaN: After parsing any leading + or -
  sign, the subroutine checks if the remaining substring is up to 8 characters
  long and matches 'INF', 'INFINITY', or 'NAN' in any letter case (e.g.,
  "-Infinity", "+inf", "NaN"). It returns ierr = 0 alongside the corresponding
  IEEE value.

  Signed Zero Preservation (-0.0): Added a signed_zero(sign_val) helper using
  ieee_negative_zero. Inputs like "-0.0", "-0e500", or negative underflows
  like "-1.0e-400" now preserve their negative sign bit.

  Allocation-Free Uppercasing: Used a fixed-length 8-character helper
  (to_upper_8) rather than deferred-length allocatable strings so the
  subroutine remains fast and free of heap allocations inside pure elemental
  array operations.

    Standard IEEE Intrinsics: Imported ieee_value, ieee_positive_inf, and
    ieee_negative_inf from ieee_arithmetic. This avoids compiler errors or
    runtime SIGFPE crashes caused by trying to compute 1.0_real64 /
    0.0_real64.

    Sign Preservation on Overflow: Added a pure elemental helper function
    signed_inf(sign_val) so inputs like "-1.80e308" return -Infinity while
    "1.80e308" returns +Infinity.

    Consistent Error 6 Handling: Updated all three overflow return paths
    (mantissa overflow, total_exp > 308, and the total_exp == 308 boundary
    check) to assign val = signed_inf(sign_val) alongside ierr = 6.

      type(unicode_type)			 :: str

DESCRIPTION
  pad(3f) pads a string with a pattern to at least the specified length. If
  the trimmed input string is longer than the requested length the trimmed

NAME
  VERIFY(3f) - [M_unicode:SEARCH] Position of a character in a string of
  characters that does not appear in a given set of characters.  (LICENSE:MIT)

SYNOPSIS
  result = verify(string, set [,back] [,kind] )

	     elemental integer function verify(string,set,back,KIND)

	      type(unicode_type),intent(in) :: string

	      type(unicode_type),intent(in) :: set
		 or
	      character(len=*),intent(in)   :: set

	      logical,intent(in),optional   :: back

CHARACTERISTICS
  •  STRING must be of type string

  •  SET must be of type string or character.

  •  BACK shall be of type logical.

  •  A default integer kind is returned.

DESCRIPTION
  VERIFY(3f) verifies that all the characters in STRING belong to the set of
  characters in SET by identifying the position of the first character in the
  string that is not in the set.

  This makes it easy to verify strings are all uppercase or lowercase, follow
  a basic syntax, only contain printable characters, and many of the
  conditions tested for with the C routines ISALNUM(3c), ISALPHA(3c),
  ISASCII(3c), ISBLANK(3c), ISCNTRL(3c), ISDIGIT(3c), ISGRAPH(3c),
  ISLOWER(3c), ISPRINT(3c), ISPUNCT(3c), ISSPACE(3c), ISUPPER(3c), and
  ISXDIGIT(3c); but for a string as well as an array of strings.

OPTIONS
  •  STRING : The string to search in for an unmatched character.

  •  SET : The set of characters that must be matched.

  •  BACK : The direction to look for an unmatched character. The left‐most
     unmatched character position isreturned unless BACK is present and
     .false., which causes the position of the right‐most unmatched character
     to be returned instead of the left‐most unmatched character.

RESULT
  If all characters of STRING are found in SET, the result is zero.

  If STRING is of zero length a zero (0) is always returned.

  Otherwise, if an unmatched character is found The position of the first or
  last (if BACK is .false.) unmatched character in STRING is returned,
  starting with position one on the left end of the string.

EXAMPLES
  Sample program I:

     use M_unicode, only : assignment(=)
     use M_unicode, only : ut=>unicode_type, ch=>character
     use M_unicode, only : write(formatted)
     use M_unicode, only : operator(==)
     use M_unicode, only : verify, replace

      & blank*(*) = " ", &

	 print *, VERIFY ("ABBA", "A")		      ! has the value 2.

     program fortran_ints
     use M_unicode, only : ut=>unicode_type,assignment(=)
     use M_unicode, only : adjustr, verify, trim, len
     use M_unicode, only : write(formatted)
     use M_unicode, only : operator(.cat.)
     use M_unicode, only : operator(==)
     implicit none
     integer :: i
     character(len=*),parameter :: asciiints(*)=[character(len=10) :: &
      "+1 ", &
      "3044848 ", &
      "30.40 ", &
      "September ", &
      "1 2 3", &
      "  -3000 ", &
      " "]
      type(ut),allocatable :: ints(:)
      if(allocated(ints))deallocate(ints)
      allocate(ints(size(asciiints))) ! gfortran bug
      ints=asciiints
      ints=trim(ints)
      ! show if strings pass or fail the test done by isint(3f)
      write(*,"('is integer?')")
      do i=1,size(ints)
	write(*,'("|",DT,T14,"|",l1,"|")') ints(i), isint(ints(i))
      enddo
      ! elemental
      write(*,"(*(g0,1x))") isint(ints)

     contains

     impure elemental function isint(line) result (lout)
     use M_unicode, only : adjustl, verify, trim
     !
     ! determine if string is a valid integer representation
     ! ignoring trailing spaces and leading spaces
     !
     character(len=*),parameter :: digits="0123456789"
     type(ut),intent(in)	:: line
     type(ut)			:: name
     logical			:: lout
	lout=.false.
	! make sure at least two characters long to simplify tests
	name=adjustl(line).cat.'  '
	! blank string
	if( name == '' )return
	! allow one leading sign
	if( verify(name%sub(1,1),ut('+‐-')) == 0 ) name=name%sub(2,len(name))
	! was just a sign
	if( name == '' )return
	lout=verify(trim(name), digits)  == 0
     end function isint

     end program fortran_ints

     impure elemental function fortran_name(line) result (lout)
     !
     ! determine if a string is a valid Fortran name
     ! ignoring trailing spaces (but not leading spaces)
     !
     character(len=*),parameter :: ints="0123456789"
     character(len=*),parameter :: lower="abcdefghijklmnopqrstuvwxyz"
     character(len=*),parameter :: upper="ABCDEFGHIJKLMNOPQRSTUVWXYZ"

     type(ut),intent(in)	:: line
     type(ut)			:: name
     logical			:: lout

	   ! first character is alphameric
	   lout = verify(name%sub(1,1), lower//upper) == 0  &
	    ! other characters are allowed in a symbol name
	    & .and. verify(name,allowed) == 0		&
	    ! allowable length
	    & .and. len(name) <= 63
	else
	   lout = .false.
	endif
     end function fortran_name

NAME
  READLINE(3f) - [M_unicode:IO] read a line from specified LUN into string up
  to line length limit (LICENSE:MIT)

SYNTAX
  function readline(lun,iostat) result(line)

      integer,intent(in),optional  :: lun
      integer,intent(out),optional :: iostat
      type(unicode_type)	   :: line

DESCRIPTION
  Read a line of any length up to programming environment maximum line length.
  Requires Fortran 2003+.

  It is primarily expected to be used when reading input which will then be
  parsed.

  The input file must have a PAD attribute of YES for the function to work
  properly, which is typically true.

  The simple use of a loop that repeatedly re-allocates a character variable
  in addition to reading the input file one buffer at a time could (depending
  on the programming environment used) be inefficient, as it could reallocate
  and allocate memory used for the output string with each buffer read.

OPTIONS
  LUN
    optional LUN (Fortran logical I/O unit) number. Defaults to stdin.

  IOSTAT
    status returned by READ(IOSTAT=IOS). If not zero, an error occurred or an
    end-of-file or end-of-record was encountered.

RETURNS
  LINE
    line read.	if IOSTAT is not zero, LINE returns the I/O error message.

EXAMPLE
  Sample program:

     program demo_readline
     use,intrinsic :: iso_fortran_env, only : stdin=>input_unit
     use,intrinsic :: iso_fortran_env, only : iostat_end
     use M_unicode, only : readline, len, trim
     use M_unicode, only : assignment(=), ch=>character, ut=>unicode_type
     implicit none
     type(ut)			  :: ln
     character(len=:),allocatable :: aline
     integer,allocatable	  :: ints(:)
     integer			  :: iostat, lun, i
     character(len=*),parameter :: filedata(*)=[character(len=80) :: &
	'The famous Confucian expression:', &
	'', &
	' "己所不欲，勿施於人"', &
	' (jǐ suǒ bù yù, wù shī yú rén)', &
	'or', &
	' "What you do not want done to yourself,', &
	' do not do to others":']
	! create a scratch file to read
	open(newunit=lun,status='scratch',pad='yes')
	write(lun,'(a)')(trim(filedata(i)),i=1,size(filedata))
	!----------------------------------------------------------------
	! read back UTF-8 byte stream and show the lines read
	rewind(unit=lun)
	do
	   ln=readline(lun,iostat=iostat)
	   if(iostat /= 0)exit
	   ! write the glyph length, byte length, line in brackets
	   write(*,'(i4,1x,i4,1x,a)')len(ln),len(ch(ln)),'['//ch(ln)//']'
	enddo
	call checkit()
	!----------------------------------------------------------------
	! the same thing except convert to default intrinsic types
	rewind(unit=lun)
	do
	   ln=readline(lun,iostat=iostat)
	   if(iostat /= 0)exit
	   ! assign the string to an allocatable array of integers
	   ints=ln
	   ! and the string to a character variable
	   aline=ln
	   write(*,'(i4,1x,i4,1x,a)')len(ln),len(aline),'['//ch(ln)//']'
	enddo
	call checkit()
	!----------------------------------------------------------------
	! again but this time show the lines as Unicode codepoints
	rewind(unit=lun)
	do
	   ln=readline(lun,iostat=iostat)
	   if(iostat /= 0)exit
	   if(len(ln) == 0)then
	      write(*,'(/,10(g0,1x)," ...")')[0,0]
	   else
	      write(*,'(/,10(g0,1x)," ...")')ln%codepoint()
	   endif
	enddo
	call checkit()
	!----------------------------------------------------------------
	! show the line as Unicode codepoints using default integer array
	rewind(unit=lun)
	do
	   ln=readline(lun,iostat=iostat)
	   if(iostat /= 0)exit
	   ! assign the string to an allocatable array of integers
	   ints=ln
	   if(size(ints) == 0)ints=[0,0]
	   write(*,'(/,10(g0,1x)," ...")')ints
	enddo
	call checkit()
	!----------------------------------------------------------------
     contains
     subroutine checkit()
	if(iostat /= iostat_end)then
	   write(*,*)'error reading input:',ch(trim(ln))
	endif
	write(*,*)
     end subroutine checkit
     end program demo_readline

regex_search(3m_unicode)			      regex_search(3m_unicode)

NAME
  regex_search(3f) - [M_unicode::REGEX] Locate strings matching a regular
  expression (LICENSE:MIT)

SYNOPSIS
  regex_search(re, text [, start_offset]) result(m)

       type(regex_t), intent(in)    :: re
       integer,intent(in)	    :: text(:)
       integer,intent(in), optional :: start_offset
       type(match_t)		    :: m

DESCRIPTION
  Searches text for the leftmost match of the compiled expression re.

  The M_unicode__regex(3) module (accessed via the M_unicode(3) module)
  compiles regular expression patterns into Nondeterministic Finite Automaton
  (NFA) bytecode and executes them over arrays of UTF-32 scalar codepoints
  (ie.`integer(int32)`arrays). Because matching operates directly on Unicode
  codepoints rather than raw UTF-8 bytes, multi-byte characters (accented
  Latin, CJK, emoji, symbols) are indexed and matched as single characters
  without splitting byte sequences.

  For convenience the procedures are generic and two other types are allowed
  -- TYPE(UNICODE_TYPE) variables and CHARACTER variables may be used whee
  CODEPOINTS(integer arrays composed of valid Unicode codepoint values).  ae
  used.

  For the CHARACTER mode the same entities are all CHARACTER variables.

  For the STRING mode all the same entities are all TYPE(UNICODE_TYPE).

  For the CODEPOINT mode all pattern compilation, text matching, and string

  replacement operate directly on arrays of UTF-8/UTF-32
    scalar codepoints (integer(int32)).

  Helper functions utoc(3) and ctou(3) convert between standard UTF-8 encoded
  Fortran character strings and codepoint arrays.

OPTIONS
  re
    Compiled regular expression. If .not. re%valid, returns m%matched =
    .false. immediately.

  text
    Subject text as an array of UTF-32 codepoints.

  start_offset
    index at which to begin searching (defaults to 1; values < 1 are clamped
    to 1).

OUTPUT
  m If m%matched is .true., the matched slice for group g (0 <= g <
    re%num_groups) is:

		       text(m%start_pos(g) : m%end_pos(g))
  whenever m%start_pos(g) >= 1 and

		       m%end_pos(g) >= m%start_pos(g).

EXAMPLE
  Sample program

     ! Demonstrates compiling a pattern with multiple capture groups,
     ! iterating through every match in a multi-line buffer using
     ! start_offset, and extracting submatches safely.
     program demo_regex_search
     use iso_fortran_env,  only : int32
     use M_unicode, only : utoc, ctou
     use M_unicode, only : regex_t, match_t
     use M_unicode, only : regex_compile, regex_search
     implicit none

     character(len=1),parameter   :: BS = achar(92)
     type(regex_t)		  :: re
     type(match_t)		  :: m
     integer(int32),allocatable   :: log_cp(:)
     integer			  :: offset, count, i
     character(len=:),allocatable :: raw_log, pat_str
     character(len=*),parameter   :: nl=new_line('a')

     raw_log = &
     '2026-11-12 08:14:02 [INFO] &
     & User "alice" connected from 10.0.1.42' &
     // nl // &
     '2026-11-12 08:14:09 [WARN] &
     & High latency (312ms) on node Zürich-01' &
     // nl // &
     ' 2026-11-12 08:14:15 [ERROR] &
     &Failed login for user "bob" from 192.168.1.99'

	! Pattern:
	!^(\d{4}-\d{2}-\d{2}\s+\d{2}:\d{2}:\d{2})\s+\[(\w+)\]\s+(.*)$
	! some compilers treat backslash as an escape character
	! so replace % with backslash
	pat_str = &
	& '^(%d{4}-%d{2}-%d{2}%s+%d{2}:%d{2}:%d{2})%s+%[(%w+)%]%s+(.*)$'
	do i=1,len(pat_str)
	   select case(pat_str(i:i))
	   case('%');pat_str(i:i)=BS
	   end select
	enddo

	re = regex_compile(utoc(pat_str))
	if (.not. re%valid) error stop 'Failed to compile log pattern'

	log_cp = utoc(raw_log)
	offset = 1
	count  = 0

	print '(A)', 'Parsed Log Entries:'
	print '(A)', &
	&'------------------------------------------------------------'

	do while (offset <= size(log_cp))
	   m = regex_search(re, log_cp, start_offset=offset)
	   if (.not. m%matched) exit

	   count = count + 1
	   print '(A,I0)', 'Entry #', count
	   print '(A)', '  Timestamp : ' // &
	   ctou(log_cp(m%start_pos(1):m%end_pos(1)))
	   print '(A)', '  Level     : ' // &
	   ctou(log_cp(m%start_pos(2):m%end_pos(2)))
	   print '(A)', '  Message   : ' // &
	   ctou(log_cp(m%start_pos(3):m%end_pos(3)))

	   offset = m%end_pos(0) + 1
	end do
     end program demo_regex_search
