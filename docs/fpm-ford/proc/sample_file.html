subroutine sample_file()
integer                                                              :: i
character(len=128),parameter :: sample_data(*)=[ CHARACTER(LEN=128) :: &
'<E><r><bo> "uni --noescape" expands C++ escape sequences to UTF-8 </bo></r></E>                                           ',&
'   \u4E03\u8EE2\u3073\u516B\u8D77\u304D\u3002                                                                             ',&
'   \u8EE2\u3093\u3067\u3082\u307E\u305F\u7ACB\u3061\u4E0A\u304C\u308B\u3002                                               ',&
'   \u304F\u3058\u3051\u305A\u306B\u524D\u3092\u5411\u3044\u3066\u6B69\u3044\u3066\u3044\u3053\u3046\u3002                 ',&
'<E><r><bo> "uni --nohtml" will convert HTML entities to actual UTF-8 Greek characters </bo></r></E>                       ',&
' The Greek alphabet consists of 24 letters, from Alpha to Omega, widely                                                   ',&
' used in mathematics, science, and engineering. The letters are                                                           ',&
'    Alpha(&Alpha;,&alpha;),Beta(&Beta;,&beta;),Gamma(&Gamma;,&gamma;),Delta(&Delta;,&delta;),Epsilon(&Epsilon;,&epsilon;),',&
'    Zeta(&Zeta;,&zeta;),Eta(&Eta;,&eta;),Theta(&Theta;,&theta;),Iota(&Iota;,&iota;),Kappa(&Kappa;,&kappa;),               ',&
'    Lambda(&Lambda;,&lambda;),Mu(&Mu;,&mu;),Nu(&Nu;,&nu;),Xi(&Xi;,&xi;),Omicron(&Omicron;,&omicron;),                     ',&
'    Pi(&Pi;,&pi;),Rho(&Rho;,&rho;),Sigma(&Sigma;,&sigma;),Tau(&Tau;,&tau;),Upsilon(&Upsilon;,&upsilon;),                  ',&
'    Phi(&Phi;,&phi;),Chi(&Chi;,&chi;),Psi(&Psi;,&psi;),Omega(&Omega;,&omega;)                                             ',&

'  #################################                                                                                       ',&
'  # Warning: proceed with caution #                                                                                       ',&
'  #################################                                                                                       ',&

'packed with umlauted ä, ö, ü, and ß (eszett):                                                                             ',&
'    <bdquo>Hinter Herrenh<ae>usern kreisen kr<ae>chzende Kr<ae>hen,                                                       ',&
'    w<ae>hrend s<ue><ss>e,                                                                                                ',&
'    m<ue>rbe G<ae>nslein gro<ss>e <Oe>lf<ae>sser umsto<ss>en.<ldquo>                                                      ',&
'to convert the sample to plain UTF-8 try                                                                                  ',&
'    <ul>uni --sample|uni --box --noescape --nohtml --nokeys | less -r</ul>                                                ',&
'to convert that UTF-8 text to all C escape sequences try                                                                  ',&
'    <ul>uni --sample|uni --box --noescape --nohtml --nokeys | uni --noescape</ul>                                         ',&
'to locate lines with non-ASCII-7 characters                                                                               ',&
'    <ul>uni --sample|uni</ul>                                                                                             ',&
'']
   write(stdout,'(a)')(trim(sample_data(i)),i=1,size(sample_data))
   stop
