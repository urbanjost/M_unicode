subroutine test_add_border()
integer                    :: length1, length2, i, j, int_1, int_2
logical                    :: bool1
type(ut),allocatable       :: textout(:), expected(:)
character(len=*),parameter :: text(*)=[character(len=15) :: &
'    WARNING, ', &
'    WARNING, ', &
' Will Robinson ']
do j=1,2
if (j == 1)then
   expected=[character(len=21*4) :: &
'┌───────────────────┐', &
'│┏━━━━━━━━━━━━━━━━━┓│', &
'│┃╔═══════════════╗┃│', &
'│┃║ Will Robinson ║┃│', &
'│┃╚═══════════════╝┃│', &
'│┗━━━━━━━━━━━━━━━━━┛│', &
'└───────────────────┘']
   textout=add_border(add_border(add_border(text(3),style='double')),style='light')
else
   expected=[character(len=21*4) :: &
'┌───────────────────┐', &
'│┏━━━━━━━━━━━━━━━━━┓│', &
'│┃╔═══════════════╗┃│', &
'│┃║    WARNING,   ║┃│', &
'│┃║    WARNING,   ║┃│', &
'│┃║ Will Robinson ║┃│', &
'│┃╚═══════════════╝┃│', &
'│┗━━━━━━━━━━━━━━━━━┛│', &
'└───────────────────┘']
   textout=add_border(add_border(add_border(text,style='double')),style='light')
endif
   expected=trim(expected)
   length1=maxval(len(textout))
   length2=maxval(len(expected))
   call check('add_border',length1 == length2,ch('length=' .cat. length1 .cat. ', expected ' .cat. length2))
   int_1=size(textout)
   int_2=size(expected)
   if( int_1 == int_2 )then
      bool1=all(textout == expected)
      call check('add_border',bool1,'compare all output text')
      if(.not.bool1)then
         write(stdout,'(*(g0,/))')(trim(textout(i)%character()),trim(expected(i)%character()),i=1,int_1)
      endif
   else
      call check('add_border',int_1 == int_2, ch('size expected ' .cat. int_2 .cat. ' got ' .cat. int_1) )
      write(stdout,'(*(g0))') 'Result:'
      write(stdout,'(g0)')(trim(textout(i)%character()),i=1,int_1)
      write(stdout,'(*(g0))') 'Expected:'
      write(stdout,'(g0)')(trim(expected(i)%character()),i=1,int_2)
   endif
enddo
end subroutine test_add_border
